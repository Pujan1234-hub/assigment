    .eq("id",shiftId);
  if(shiftError){alert("Could not end shift: "+shiftError.message);return}

  // by open shifts, so a successful shift end removes the staff member immediately.

};

async function locationPermissionState(){
  try{
    if(navigator.permissions){
      const s=await navigator.permissions.query({name:"geolocation"});
      return s.state;
    }
  }catch(e){}
  return "unknown";
}
async function startGps(){
  if(!navigator.geolocation){gpsError("GPS not supported.");return}
  const perm=await locationPermissionState();
  // Important: permission is only triggered by START SHIFT. We never ask on page load.
  watchId=navigator.geolocation.watchPosition(async p=>{
      latestPos=p;
      updateGpsUi(p);
      if(!activeShift)return;
      const now=Date.now();
      const moved=(lastPublishedLat==null)?99999:distanceMeters(
        lastPublishedLat,lastPublishedLon,p.coords.latitude,p.coords.longitude
      );
      // Fast when moving, calm when stationary.
      if(moved>=2 || now-lastGpsPublishAt >= (cfg.LOCATION_UPDATE_MS||1500)){
        await publishLocation(p);
        lastGpsPublishAt=Date.now();
        lastPublishedLat=p.coords.latitude;
        lastPublishedLon=p.coords.longitude;
      }
    }, e=>gpsError(e.message),
    {enableHighAccuracy:true,maximumAge:0,timeout:10000}
  );
  timer=setInterval(()=>{
    if(latestPos&&activeShift&&Date.now()-lastGpsPublishAt >= (cfg.LOCATION_HEARTBEAT_MS||2000)){
      publishLocation(latestPos);
      lastGpsPublishAt=Date.now();
    }
  },1000);
}
function stopGps(){
  if(watchId!==null){navigator.geolocation.clearWatch(watchId);watchId=null}
  if(timer){clearInterval(timer);timer=null}
  latestPos=null;
}
function updateGpsUi(p){
  const acc=Math.round(p.coords.accuracy||0);
  $("gpsSub").textContent=`Location active • accuracy ±${acc} m`;
  $("fTime").textContent=fmtTime(new Date().toISOString());
  const effectiveArea=effectiveAreaFromPosition(p);
  $("fArea").textContent=effectiveArea;renderLiveStatusTask();
}
async function publishLocation(p){
  if(!activeShift||!profile)return;
  const now=new Date().toISOString();
  const payload={
    staff_id:profile.id,shift_id:activeShift.id,
    latitude:p.coords.latitude,longitude:p.coords.longitude,
    accuracy_m:Math.round(p.coords.accuracy||0),
    area:effectiveAreaFromPosition(p),task:$("task").value.trim()||null,is_on_shift:true,updated_at:now
  };
  const {error}=await sb.from("current_locations").upsert(payload,{onConflict:"staff_id"});
  if(!error){
    $("fTime").textContent=fmtTime(now);
    if(firstGpsForShift){
      firstGpsForShift=false;
      sb.from("shifts").update({start_area:effectiveAreaFromPosition(p)}).eq("id",activeShift.id);
    }
  }
}

};
window.setStaffPin=async(staffId)=>{
  if(!profile||profile.access_level!=="admin")return;

async function loadTaskInbox(){
  if(!profile||staffTaskLoadBusy)return;
  staffTaskLoadBusy=true;
  const {data,error}=await sb.from("task_assignments")
    .select("*")
    .eq("staff_id",profile.id)
    .in("status",["assigned","acknowledged"])
    .order("created_at",{ascending:false})
    .limit(20);
  if(error){staffTaskLoadBusy=false;return;}
  taskInboxRows=data||[];
  currentAssignment=taskInboxRows[0]||null;
  renderTaskInbox();
  renderCurrentAssignment();
  staffTaskLoadBusy=false;
}
function renderLiveStatusTask(){
  const el=$("fTask");
  if(!el)return;
  const t=currentAssignment;
  if(!t){
    const manual=($("task")?.value||"").trim();
    el.innerHTML=manual?`<div class="live-task-text">${escapeHtml(manual)}</div>`:`<span class="live-task-empty">—</span>`;
    return;
  }
  const status=(t.status||"assigned").toLowerCase();
  const ack=status==="assigned"?`<button class="task-btn primary" onclick="setTaskState('${escapeAttr(t.id)}','acknowledged')">ACK</button>`:"";
  el.innerHTML=`
    <div class="live-task-top">
      <span class="live-task-pill ${status}">${escapeHtml(status.toUpperCase())}</span>
      <div class="live-task-text">${escapeHtml(t.task_text||"Task")}</div>
    </div>
    <div class="live-task-meta">${escapeHtml(t.area||"Control Room")} • ${escapeHtml((t.priority||"normal").toUpperCase())}</div>
    <div class="live-task-actions">
      ${ack}
      <button class="task-btn done" onclick="setTaskState('${escapeAttr(t.id)}','completed')">DONE</button>
      <button class="task-btn skip" onclick="setTaskState('${escapeAttr(t.id)}','declined')">SKIP</button>
    </div>`;
}
function renderCurrentAssignment(){
  const box=$("assignedTaskBox");
  const manualInput=$("task");
  renderLiveStatusTask();
  if(!box)return;
  if(!currentAssignment){
    box.classList.add("hidden");
    if(manualInput)manualInput.classList.remove("hidden");
    return;
  }
  if(manualInput)manualInput.classList.add("hidden");
  box.classList.remove("hidden");
  const status=(currentAssignment.status||"assigned").toLowerCase();
  $("assignedTaskText").textContent=currentAssignment.task_text||"—";
  $("assignedTaskMeta").textContent=`${currentAssignment.area||"Control Room"} • ${(currentAssignment.priority||"normal").toUpperCase()}`;
  const statusEl=$("assignedTaskStatus");
  if(statusEl){statusEl.textContent=status.toUpperCase();statusEl.className=`live-task-pill ${status}`}
  $("ackTaskBtn").classList.toggle("hidden",status!=="assigned");
}
