function distanceMeters(lat1,lon1,lat2,lon2){
  const R=6371000, toRad=x=>x*Math.PI/180;
  const dLat=toRad(lat2-lat1), dLon=toRad(lon2-lon1);
  const a=Math.sin(dLat/2)**2+Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLon/2)**2;
  return 2*R*Math.atan2(Math.sqrt(a),Math.sqrt(1-a));
}

async function staffPinSignIn(){
  if(!configured){msg($("staffPinLoginMsg"),"Connect Supabase first.");return}
  const staffCode=$("staffCode").value.trim();
  const pin=$("staffPin").value.trim();
  if(!staffCode || !/^\d{4}$/.test(pin)){
    msg($("staffPinLoginMsg"),"Enter your Staff ID and 4-digit PIN.");return;
  }

  $("staffPinLoginBtn").disabled=true;
  msg($("staffPinLoginMsg"),"Checking staff access…",true);

  // Staff PIN sessions use a low-privilege anonymous Supabase identity behind the scenes.
  // If another account is already signed in on this browser, sign it out before claiming staff mode.
  let {data:{session}}=await sb.auth.getSession();
  if(session){
    const {data:{user}}=await sb.auth.getUser();
    if(!user?.is_anonymous){
      await sb.auth.signOut();
      session=null;
    }
  }

  if(!session){
    const {error:anonErr}=await sb.auth.signInAnonymously();
    if(anonErr){
      $("staffPinLoginBtn").disabled=false;
      msg($("staffPinLoginMsg"),"Staff PIN login needs Anonymous Sign-Ins enabled in Supabase Auth.");
      return;
    }
  }

  const deviceName=(navigator.userAgent||"Mobile browser").slice(0,120);
  const {error}=await sb.rpc("claim_staff_with_pin",{
    p_staff_code:staffCode,
    p_pin:pin,
    p_device_name:deviceName
  });

  if(error){
    await sb.auth.signOut();
    $("staffPinLoginBtn").disabled=false;
    msg($("staffPinLoginMsg"),error.message||"Staff ID or PIN not recognised.");
    return;
  }

  $("staffPinLoginBtn").disabled=false;
  $("staffPin").value="";
  await loadProfile();
}
